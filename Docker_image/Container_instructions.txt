# when you create a new container from this image, run below commands to setup your camera


# 1. uninstall the below libraries

pip uninstall opencv-contrib-python -y
pip uninstall opencv-python -y

# 2. Camera Configuration for imx219 camera 

sudo /opt/nvidia/jetson-io/jetson-io.py

# A window will open after the above command, Go to Configure Jetson 22-pin Header -> Configure for compatible hardware.
# then select camera IMX219-A 

# 3. Restart nvargus which is responsidble for camera

sudo systemctl restart nvargus-daemon

# 4. Logout and login to your system


# 5. test your camera feed

gst-launch-1.0 nvarguscamerasrc sensor-id=0 ! 'video/x-raw(memory:NVMM),width=1920,height=1080,format=NV12,framerate=30/1' ! nvvidconv ! xvimagesink
